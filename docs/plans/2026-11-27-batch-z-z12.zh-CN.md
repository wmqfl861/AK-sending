```dart
typedef _GrantRow = ({
  String uri,
  bool read,
  bool write,
  bool tree,
  int persistedTimeMs,
});
typedef _GrantTable = Map<_GrantRow, int>;

_GrantTable? _grantTableOf(Object? value) {
  if (value is! List) return null;
  final table = <_GrantRow, int>{};

  for (final item in value) {
    if (item is! Map || item.length != 5) return null;
    final Object? uri = item['uri'];
    final Object? read = item['read'];
    final Object? write = item['write'];
    final Object? tree = item['tree'];
    final Object? time = item['persistedTimeMs'];
    if (uri is! String || uri.isEmpty || read is! bool ||
        write is! bool || tree is! bool || time is! int) {
      return null;
    }

    final row = (
      uri: uri, read: read, write: write, tree: tree, persistedTimeMs: time,
    );
    table[row] = (table[row] ?? 0) + 1;
  }
  return table;
}

String _grantComparison(_GrantTable? before, _GrantTable? after) {
  if (before == null) return 'before-invalid';
  if (after == null) return 'after-invalid';
  if (before.length != after.length) return 'changed';
  for (final entry in before.entries) {
    if (after[entry.key] != entry.value) return 'changed';
  }
  return 'same';
}

Future<_GrantTable?> _readGrantTable() async {
  final answer = await _call(
    _storageChannel.invokeMethod<Object?>('listSafPersisted'),
  );
  if (!answer.answered || answer.error != null) return null;
  return _grantTableOf(answer.value);
}

void _checkGrantTable(List<String> failures, String part, String point,
    _GrantTable? before, _GrantTable? after) {
  final result = _grantComparison(before, after);
  zSay('$part|GRANT|point=$point|result=$result');
  if (result != 'same') failures.add('$point-grants-$result');
}
```
